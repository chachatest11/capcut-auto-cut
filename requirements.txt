#  라이브러리는 계속 업데이트되므로, 설치 중 충돌이 나면 해당 줄의
#  '==버전' 부분을 지우고 다시 설치해 최신 호환 버전을 받으면 됩니다.

# --- GUI (모던 다크 테마 데스크톱 UI) ---
#  Windows 공식 파이썬에는 tkinter가 기본 포함되어 별도 설치가 필요 없습니다.
customtkinter==5.2.2

# --- VAD: 사람 목소리 구간 감지용 Silero-VAD ONNX 추론 엔진 ---
#  silero_vad.onnx 모델 파일을 직접 onnxruntime 으로 돌립니다.
onnxruntime==1.20.1

# --- STT: 음성을 실제 자막 텍스트로 받아쓰기 (Whisper 경량/고속 구현) ---
#  내부적으로 ctranslate2 백엔드를 사용합니다(자동 설치됨).
faster-whisper==1.1.0

# --- FFmpeg 바이너리 제공 (영상 오디오 추출 / 컷 / 재합성) ---
#  설치 시 Windows용 정적 ffmpeg.exe 실행파일을 함께 내려받아 줍니다.
#  나중에 --onefile 빌드에 이 바이너리를 통째로 내장합니다.
imageio-ffmpeg==0.5.1

# --- 오디오 수치 연산 (PCM 샘플 배열 처리) ---
#  numpy 2.x 는 일부 백엔드와 충돌 사례가 있어 1.x 로 고정합니다.
numpy==1.26.4

# --- 패키징 (4단계에서 사용, .exe 단일 실행 파일 빌드) ---
pyinstaller==6.11.1
